document.addEventListener(“DOMContentLoaded”, function () { const
ageRadios =
document.querySelectorAll(‘input[type=“radio”][name=“ageGroup”]’); const
genreCheckboxes =
document.querySelectorAll(‘input[type=“checkbox”].genre-checkbox’);

    const ageDisplay = document.getElementById("ageDisplay");
    const genreCount = document.getElementById("genreCount");
    const totalGenres = document.getElementById("totalGenres");
    const genreList = document.getElementById("genreList");

    const pollSubmitBtn = document.getElementById("pollSubmitBtn");
    const pollResetBtn = document.getElementById("pollResetBtn");

    totalGenres.textContent = genreCheckboxes.length;

    for (let i = 0; i < ageRadios.length; i++) {
        ageRadios[i].addEventListener("change", function () {
            ageDisplay.textContent = this.value;
        });
    }

    pollSubmitBtn.addEventListener("click", function () {
        let selectedCount = 0;

        for (let i = 0; i < genreCheckboxes.length; i++) {
            if (genreCheckboxes[i].checked) {
                selectedCount++;
            }
        }

        let selectedAge = "";
        let radioIndex = 0;

        while (radioIndex < ageRadios.length) {
            if (ageRadios[radioIndex].checked) {
                selectedAge = ageRadios[radioIndex].value;
                break;
            }
            radioIndex++;
        }

        let selectedGenres = [];
        let genreIndex = 0;

        if (genreCheckboxes.length > 0) {
            do {
                if (genreCheckboxes[genreIndex].checked) {
                    selectedGenres.push(genreCheckboxes[genreIndex].value);
                }

                genreIndex++;
            } while (genreIndex < genreCheckboxes.length);
        }

        ageDisplay.textContent = selectedAge || "None";
        genreCount.textContent = selectedCount;

        if (selectedGenres.length > 0) {
            genreList.textContent = selectedGenres.join(", ");
        } else {
            genreList.textContent = "None";
        }
    });

    pollResetBtn.addEventListener("click", function () {
        for (let i = 0; i < genreCheckboxes.length; i++) {
            genreCheckboxes[i].checked = false;
        }

        genreCount.textContent = "0";
        genreList.textContent = "None";
    });

});
